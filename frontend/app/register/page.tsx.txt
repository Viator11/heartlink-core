"use client";
import { useState } from "react";

export default function Register(){
  const [email,setEmail]=useState(""); const [password,setPassword]=useState("");
  const [msg,setMsg]=useState("");

  const handleRegister=async()=>{
    const r=await fetch(`${process.env.NEXT_PUBLIC_API_URL}/auth/register`,{
      method:"POST",
      headers:{"Content-Type":"application/json"},
      body:JSON.stringify({email,password})
    });
    const data=await r.json();
    setMsg(r.ok?"✅ Registered! Check email":"❌ "+(data.detail||"Error"));
  };

  return (
    <div className="flex flex-col gap-3">
      <h2 className="text-xl font-bold">Register</h2>
      <input className="border p-2" placeholder="Email" value={email} onChange={e=>setEmail(e.target.value)} />
      <input className="border p-2" placeholder="Password" type="password" value={password} onChange={e=>setPassword(e.target.value)} />
      <button onClick={handleRegister} className="bg-green-600 text-white p-2 rounded">Register</button>
      {msg && <p className="text-sm text-gray-600">{msg}</p>}
    </div>
  );
}
