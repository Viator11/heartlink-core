"use client";
import { useState } from "react";

export default function Login() {
  const [email,setEmail]=useState("");
  const [password,setPassword]=useState("");
  const [msg,setMsg]=useState("");

  const handleLogin=async()=>{
    try{
      const r=await fetch(`${process.env.NEXT_PUBLIC_API_URL}/auth/login`,{
        method:"POST",
        headers:{"Content-Type":"application/json"},
        body:JSON.stringify({email,password})
      });
      const data=await r.json();
      if(r.ok){
        localStorage.setItem("token",data.access.token);
        setMsg("✅ Logged in!");
      } else setMsg(data.detail||"Error");
    }catch(e){setMsg("Network error");}
  };

  return (
    <div className="flex flex-col gap-3">
      <h2 className="text-xl font-bold">Login</h2>
      <input className="border p-2" placeholder="Email" value={email} onChange={e=>setEmail(e.target.value)} />
      <input className="border p-2" placeholder="Password" type="password" value={password} onChange={e=>setPassword(e.target.value)} />
      <button onClick={handleLogin} className="bg-blue-600 text-white p-2 rounded">Login</button>
      {msg && <p className="text-sm text-gray-600">{msg}</p>}
    </div>
  );
}
